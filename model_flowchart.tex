\documentclass[tikz, border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, backgrounds}

% ── color palette ──────────────────────────────────────────────────────────────
\definecolor{inputcol}{RGB}{220,235,252}
\definecolor{enccol} {RGB}{198,230,210}
\definecolor{gcncol}  {RGB}{255,236,179}
\definecolor{poolcol} {RGB}{230,210,250}
\definecolor{headcol} {RGB}{255,210,210}
\definecolor{outcol}  {RGB}{200,230,230}
\definecolor{boxbg}   {RGB}{245,245,245}

% ── node styles ────────────────────────────────────────────────────────────────
\tikzset{
  base/.style   = {rectangle, rounded corners=4pt, minimum width=5.2cm,
                   minimum height=0.75cm, align=center, font=\small},
  input/.style  = {base, fill=inputcol, draw=blue!40},
  enc/.style    = {base, fill=enccol,   draw=green!50!black},
  gcn/.style    = {base, fill=gcncol,   draw=orange!70, minimum width=6.4cm},
  pool/.style   = {base, fill=poolcol,  draw=violet!60},
  head/.style   = {base, fill=headcol,  draw=red!50,
                   minimum width=2.0cm, minimum height=0.65cm},
  out/.style    = {base, fill=outcol,   draw=teal!60,
                   minimum width=2.0cm, minimum height=0.65cm},
  arr/.style    = {-{Stealth[length=5pt]}, thick},
  groupbox/.style = {rectangle, rounded corners=6pt, draw=gray!50,
                     dashed, fill=boxbg, inner sep=8pt},
  label/.style  = {font=\footnotesize\itshape, text=gray!70}
}

\begin{document}
\begin{tikzpicture}[node distance=0.55cm]

% ── (0) Input ──────────────────────────────────────────────────────────────────
\node[input] (x)     {$\mathbf{x}$: node type \& attribute};
\node[input, right=0.4cm of x] (dep) {node\_depth};
\node[input, below=0.3cm of x, xshift=1.5cm] (ea)
      {edge\_index, edge\_attr $\in\mathbb{R}^{E\times2}$};

% ── (1) ASTNodeEncoder ────────────────────────────────────────────────────────
\node[enc, below=0.55cm of ea] (enc)
      {\textbf{ASTNodeEncoder}\\
       type-emb $+$ attr-emb $+$ depth-emb};

\node[label, left=0.15cm of enc] {emb\_dim=300};

% ── (2) GCN Layers ────────────────────────────────────────────────────────────
\node[gcn, below=0.6cm of enc] (gcn1)
      {\textbf{GCNConv} + BN + ReLU + Dropout(0.5)};
\node[gcn, below=0.45cm of gcn1] (gcn2)
      {\textbf{GCNConv} + BN + ReLU + Dropout(0.5)};
\node[gcn, below=0.45cm of gcn2] (gcn3)
      {\textbf{GCNConv} + BN \hspace{0.5em}[\,no ReLU\,] + Dropout(0.5)};

\node[label, right=0.1cm of gcn1] {layer 1};
\node[label, right=0.1cm of gcn2] {layer 2};
\node[label, right=0.1cm of gcn3] {layer 3};

% dashed box around GCN layers
\begin{scope}[on background layer]
  \node[groupbox, fit=(gcn1)(gcn2)(gcn3),
        label={[font=\small\bfseries, text=orange!80!black]above:GCN Backbone ($\times3$)}] (gcnbox) {};
\end{scope}

% ── GCNConv detail (side annotation) ─────────────────────────────────────────
\node[font=\scriptsize, align=left, right=0.55cm of gcnbox, text=gray!80,
      yshift=0.2cm]
      (detail)
      {$\text{msg} = \sigma(x_j + \text{EdgeEnc}(e_{ij}))$\\
       $\cdot\, D^{-\tfrac{1}{2}}_{ii}\,D^{-\tfrac{1}{2}}_{jj}$\\[2pt]
       $h_i = \text{agg} + \tfrac{\text{ReLU}(x_i + r)}{\deg_i}$};

% ── (3) Global Mean Pooling ───────────────────────────────────────────────────
\node[pool, below=0.6cm of gcnbox] (pool)
      {\textbf{Global Mean Pooling}\\
       $h_{\mathcal{G}} = \frac{1}{|V|}\sum_{i} h_i$};

\node[label, left=0.15cm of pool] {$\mathbb{R}^{300}$};

% ── (4) Prediction Heads ──────────────────────────────────────────────────────
\node[head, below=0.65cm of pool, xshift=-4.0cm] (h1) {Linear\\pos 1};
\node[head, right=0.25cm of h1]                  (h2) {Linear\\pos 2};
\node[head, right=0.25cm of h2]                  (h3) {Linear\\pos 3};
\node[head, right=0.25cm of h3]                  (h4) {Linear\\pos 4};
\node[head, right=0.25cm of h4]                  (h5) {Linear\\pos 5};

\node[label, above=0.05cm of h1, xshift=2.1cm]
      {$5\times$ Linear($300 \to |\mathcal{V}|$)};

% dashed box around heads
\begin{scope}[on background layer]
  \node[groupbox, fit=(h1)(h2)(h3)(h4)(h5),
        label={[font=\small\bfseries, text=red!70]above:Prediction Heads}] (headbox) {};
\end{scope}

% ── (5) Output ────────────────────────────────────────────────────────────────
\node[out, below=0.6cm of h1] (o1) {$\hat{y}_1$};
\node[out, below=0.6cm of h2] (o2) {$\hat{y}_2$};
\node[out, below=0.6cm of h3] (o3) {$\hat{y}_3$};
\node[out, below=0.6cm of h4] (o4) {$\hat{y}_4$};
\node[out, below=0.6cm of h5] (o5) {$\hat{y}_5$};

\node[label, below=0.15cm of o3, xshift=0.0cm]
      {Predicted code token sequence (F1 evaluated)};

% ── Arrows ────────────────────────────────────────────────────────────────────
% inputs -> encoder
\draw[arr] (x.south)  -- ++(0,-0.22) -| (enc.north);
\draw[arr] (dep.south) -- ++(0,-0.3)  -| (enc.north);
\draw[arr] (ea.south) -- (enc.north);

% encoder -> gcn1
\draw[arr] (enc.south) -- (gcn1.north);

% gcn layers
\draw[arr] (gcn1.south) -- (gcn2.north);
\draw[arr] (gcn2.south) -- (gcn3.north);

% gcn3 -> pool
\draw[arr] (gcn3.south) -- ++(0,-0.22) -- (pool.north);

% pool -> each head
\foreach \h in {h1,h2,h3,h4,h5}{
  \draw[arr] (pool.south) -- ++(0,-0.22) -| (\h.north);
}

% heads -> outputs
\draw[arr] (h1.south) -- (o1.north);
\draw[arr] (h2.south) -- (o2.north);
\draw[arr] (h3.south) -- (o3.north);
\draw[arr] (h4.south) -- (o4.north);
\draw[arr] (h5.south) -- (o5.north);

\end{tikzpicture}
\end{document}
